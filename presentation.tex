% ================================================ BACKGROUND (4 frames)

\begin{frame}{An autoresearch loop runs its own experiments}
  \begin{itemize}
    \item The model is given a goal, code it may edit, and an executor
    \item The executor runs one trial and returns a single scalar
    \item From there it proceeds without a human in the loop
  \end{itemize}
  \vfill
  \begin{block}{The cycle}
    Propose a change, execute it, read the outcome, decide what to attempt
    next, repeat until the budget is exhausted.
  \end{block}
  \vfill
  \begin{itemize}
    \item Progress is enforced by version control: commit, train, keep if the
          score improved, otherwise \texttt{git reset}
    \item The repository only ever advances
  \end{itemize}
\end{frame}

\begin{frame}{Why this matters now}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Karpathy's \texttt{autoresearch}, March 2026
        \item Forked several thousand times since
        \item It is what fixed the current meaning of the term
        \item People are running it against production systems
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Shopify Liquid}
        \small
        About 120 experiments\\
        $53\%$ faster parse and render\\
        Author's own verdict: probably overfitted\\
        Never merged
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{alertblock}{The uncomfortable part}
    The loops demonstrably function. The reasons for their behaviour are
    unknown, including to the people running them.
  \end{alertblock}
\end{frame}

\begin{frame}{The architecture is three files, and one of them is the controller}
  \begin{center}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{File} & \textbf{Holds} & \textbf{Edited by}\\
    \midrule
    \texttt{prepare.py} & the data and the score          & nobody\\
    \texttt{train.py}   & the model and the training loop & the agent\\
    \texttt{\prog}      & the instructions                & the human\\
    \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  \begin{itemize}
    \item \texttt{prepare.py} is unreachable, so every experiment is scored
          identically
    \item No scheduler, no driver program, no orchestration code of any kind
    \item The nine step experiment cycle exists only as English prose inside
          \texttt{\prog}, and the model itself is the execution layer
  \end{itemize}
  \vfill
  \begin{block}{The observation the project rests on}
    \texttt{\prog} is not configuration supplied to a controller. It \emph{is}
    the controller. A controller written in code can be read. This one cannot.
  \end{block}
\end{frame}

% ========================================== RESEARCH QUESTION (1 frame)

\begin{frame}{Research questions}
  \begin{block}{RQ1}
    Which components of \texttt{\prog} causally determine which aspects of the
    loop's research \emph{process}, and how large is each effect?
  \end{block}
  \vfill
  \begin{block}{RQ2}
    Do the components act independently of one another, or does the effect of
    one depend on the level of another?
  \end{block}
  \vfill
  \begin{itemize}
    \item The contribution is a method, not an architecture. The aim is not a
          better loop, but an account of what an existing one responds to
    \item Scope: no human baseline and no external standard of good practice.
          Every comparison is between two configurations of the same system
  \end{itemize}
\end{frame}

% ===================================== EXPERIMENTAL DESIGN (9 frames)

\begin{frame}{From prose to a configuration vector}
  \begin{itemize}
    \item One narrowing first: the search space is a hyperparameter space, not
          arbitrary code, so the instruction file stays the only free variable
    \item Every file is rendered from one template
    \item The fixed part gives the research question, the executor interface
          and the JSON response format, byte identical across all 32 files
    \item The variable part is five paragraphs, each written at one of two
          levels. Nothing else in the file varies
  \end{itemize}
  \vfill
  \begin{block}{A file becomes a point in a space}
    \centering
    $c = (c_1,\ldots,c_5), \qquad c_i \in \{0,1\}
     \qquad\Longrightarrow\qquad 2^5 = 32$ files
  \end{block}
\end{frame}

\begin{frame}{The five components: the independent variables}
  \scriptsize
  \begin{tabular}{@{}clp{4.3cm}p{4.3cm}@{}}
    \toprule
    & \textbf{Component} & \textbf{Level 0} & \textbf{Level 1}\\
    \midrule
    $M$ & Evaluation criterion &
      Success left undefined; the loop judges quality as it sees fit. &
      Success is 5 fold cross validated accuracy; the standard deviation breaks
      ties only.\\[2pt]
    $B$ & Search breadth &
      Commit to one model family for the session. &
      Compare both families and vary several hyperparameters at once.\\[2pt]
    $S$ & Stopping rule &
      Fixed budget of exactly three experiments. &
      Adaptive budget up to eight, halt when improvement falls below $0.002$.\\[2pt]
    $O$ & Reporting style &
      Terse: the action taken and the numbers involved. &
      Full sentences: hypothesis, expectation, interpretation.\\[2pt]
    $E$ & Search emphasis &
      Explore first: several configurations before refining any. &
      Exploit first: refine a promising configuration immediately.\\
    \bottomrule
  \end{tabular}
  \vfill
  \fnote{These categories were not invented for the study. A production
        \texttt{\prog} hardcodes a baseline to exceed, specifies failure
        handling, fixes a reporting discipline and imposes a parsimony rule.}
\end{frame}

\begin{frame}[fragile]{Every file is rendered from this template}
\begin{lstlisting}[style=tpl]
# Research Task: {{ task_title }}                                    <- set by the task, not by c

You are an autonomous research agent. Your task is to investigate the following
question through real, executed experiments (not simulated ones):

> {{ research_question }}                                            <- set by the task, not by c

You have access to a fixed dataset ({{ dataset_name }}, already loaded and split
into train/validation) and to a tool that fits either a RandomForestClassifier or
a GradientBoostingClassifier with hyperparameters you choose. The tool always
returns the same record: 5-fold cross-validated accuracy on the training split,
its standard deviation across folds, and accuracy on the held-out validation split.

## Evaluation
{{ metric_text }}                                                    (*\textcolor{accent}{\bfseries <== slot M}*)

## Search strategy
{{ breadth_text }}                                                   (*\textcolor{accent}{\bfseries <== slot B}*)
{{ emphasis_text }}                                                  (*\textcolor{accent}{\bfseries <== slot E}*)

## Budget and stopping
{{ stopping_text }}                                                  (*\textcolor{accent}{\bfseries <== slot S}*)

## Reporting style
{{ output_format_text }}                                             (*\textcolor{accent}{\bfseries <== slot O}*)

## Required response format
At every step, respond with a single JSON object and nothing else, matching one of these two shapes:
{"action": "propose", "hypothesis": ..., "model": "random_forest" | "gradient_boosting", "params": {...}}
{"action": "interpret", "interpretation": ..., "decision": "continue" | "stop", "final_recommendation": ...}
\end{lstlisting}
  \vspace{-0.1cm}
  \fnote{Everything outside the five marked slots is byte identical across all
        32 files of a task. The three remaining substitutions are set by the
        dataset, so they are constant within a task and differ only between
        tasks.}
\end{frame}

\begin{frame}[fragile]{The ten paragraphs, verbatim (1 of 2)}
\begin{lstlisting}[style=para]
(*\textcolor{accent}{\bfseries slot M  Evaluation criterion}*)
 [0] Evaluate model quality in whatever way you judge appropriate.
 [1] Evaluate every model using 5-fold cross-validated accuracy on the training split, AND report
     the standard deviation across folds as a measure of stability. A configuration is only better
     than another if its mean accuracy is higher; use the standard deviation only to break ties or
     to flag instability.

(*\textcolor{accent}{\bfseries slot B  Search breadth}*)
 [0] Stay within a single model family per experiment. Pick either RandomForestClassifier or
     GradientBoostingClassifier for the whole session unless you have exhausted reasonable
     hyperparameters for it and explicitly decide to switch.
 [1] You are free to compare RandomForestClassifier and GradientBoostingClassifier side by side
     within the same session, and to vary multiple hyperparameters of each at once.

(*\textcolor{accent}{\bfseries slot S  Stopping rule}*)
 [0] You have a fixed budget of exactly 3 experiments in total. Run exactly 3, then stop and report
     your final recommendation, regardless of the outcome of experiment 3.
 [1] You have a budget of up to 8 experiments. After each experiment, decide whether to continue:
     stop as soon as an experiment fails to improve the evaluation metric (defined above) by more
     than 0.002 over the best result so far, or when you reach 8 experiments, whichever comes first.
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{The ten paragraphs, verbatim (2 of 2)}
\begin{lstlisting}[style=para]
(*\textcolor{accent}{\bfseries slot O  Reporting style}*)
 [0] Keep every response short: state the action you are taking and the numbers involved, with
     minimal prose.
 [1] Explain your reasoning at every step in full sentences: what you hypothesize, why, what you
     expect to observe, and how you interpret the actual result relative to that expectation.

(*\textcolor{accent}{\bfseries slot E  Search emphasis}*)
 [0] Prioritize breadth of search: try several meaningfully different configurations before you
     spend more than one experiment refining any single one of them.
 [1] Prioritize depth: as soon as one configuration looks promising, immediately refine it with
     nearby variations (e.g. slightly different hyperparameter values) before trying anything
     unrelated.
\end{lstlisting}
  \vfill
  \begin{block}{What a configuration is, concretely}
    \small
    $c = (1,0,1,1,0)$ is the file \texttt{M1-B0-S1-O1-E0}: it takes paragraph
    \texttt{[1]} in the evaluation slot, \texttt{[0]} in breadth, \texttt{[1]}
    in stopping, \texttt{[1]} in reporting and \texttt{[0]} in emphasis. Its
    neighbour \texttt{M1-B0-S1-O0-E0} is the same file with one paragraph
    swapped and nothing else changed. The other 30 files are the remaining
    combinations.
  \end{block}
\end{frame}

\begin{frame}{The complete factorial makes interactions estimable}
  \begin{itemize}
    \item With all 32 files, every pairwise interaction $\beta_{ij}$ is
          estimated separately from every main effect
    \item The independence of the components therefore becomes a question
          answerable \emph{from the data}, rather than one the design has to
          assume
  \end{itemize}
  \vfill
  \begin{block}{This is what makes RQ2 testable}
    Does the effect of one component depend on the level of another, or do they
    simply add?
  \end{block}
\end{frame}

\begin{frame}{Tasks, executor and controls}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Breast Cancer Wisconsin, $569\times30$, two classes
        \item Wine, $178\times13$, three classes
        \item One stratified $70/30$ split per task, same seed throughout
        \item Two model families under a fixed hyperparameter whitelist
        \item Baseline $a_0(d)$: a default fit, better of the two families,
              computed with no language model. $0.960$ on both tasks
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Held fixed, so they explain nothing}
        \scriptsize
        agent model pinned to \texttt{gpt-4o-mini-2024-07-18}, the split and its
        seed, the executor and whitelist, the fixed template, the response
        format, the harness caps. Runs are separate conversations.
      \end{block}
      \vspace{-0.15cm}
      \begin{alertblock}{Temperature $=0.7$, not $0$}
        \scriptsize
        Run to run variance \emph{is} the error term every coefficient is tested
        against. Greedy decoding would drive it to zero and every standard error
        with it.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \fnote{The executor returns the same record every time: cross validated
        accuracy, its across fold standard deviation, held out validation
        accuracy. The instruction file alters what the loop attends to, not what
        is measured. Every run passes an explicit sampling seed and every cell
        draws from the same seed list, so the draws cannot favour one
        configuration.}
\end{frame}

\begin{frame}{Design, budget and metrics}
  \begin{block}{}
    \centering
    $N \;=\; \underbrace{2^5}_{\text{configurations}} \times
             \underbrace{D}_{\text{2 tasks}} \times
             \underbrace{R}_{\text{20 replicates}} \;=\; 1{,}280$ runs,
             64 cells, 20 each
  \end{block}
  \vspace{-0.1cm}
  \scriptsize
  \begin{tabular}{@{}llp{6.9cm}@{}}
    \toprule
    & \textbf{Metric} & \textbf{Definition}\\
    \midrule
    Outcome    & Gain over default  & recommended score minus $a_0(d)$; negative means worse than untuned defaults\\
    Outcome    & Improvement rate   & share of executed trials beating the best seen earlier in the run\\
    Efficiency & Wasted trial ratio & (rejected or malformed $+$ exact duplicates) over proposals attempted\\
    Efficiency & Cost to best       & executor calls up to and including the run's best score\\
    \bottomrule
  \end{tabular}
  \normalsize
  \vfill
  \begin{itemize}
    \item $R=20$ fixes the smallest detectable level switch at $0.22\,\sigma$
          per dataset
    \item Every metric is read off the structured transcript. No judge model,
          no scoring library, so nothing in the measurement can drift with the
          instruction being tested
    \item All metrics computed over the \textbf{first three experiments}, the
          number available in every run whatever the stopping rule
  \end{itemize}
\end{frame}

\begin{frame}{The regression model}
  \begin{block}{Orthogonal coding, then main effects with the task controlled}
    \centering
    $X_i = +1$ at level 1, \quad $X_i = -1$ at level 0\\[4pt]
    $Y \;=\; \beta_0 + \sum_{i=1}^{5}\beta_i X_i
       + \sum_{j=2}^{D}\gamma_j D_j + \varepsilon$
  \end{block}
  \vspace{-0.1cm}
  \begin{block}{Interactions, free because all 32 cells were run}
    \centering
    $Y \;=\; \beta_0 + \sum_{i}\beta_i X_i
       + \sum_{i<j}\beta_{ij}X_iX_j
       + \sum_{j}\gamma_j D_j + \varepsilon$
  \end{block}
  \vspace{-0.1cm}
  \begin{itemize}
    \item $2\beta_i$ is the average change in $Y$ from switching component $i$,
          and is what is reported throughout
    \item $\gamma_j$ absorbs task difficulty and headroom above $a_0$, which
          would otherwise inflate every standard error
    \item OLS on individual runs, HC3 standard errors, Benjamini and Hochberg
          adjustment within each metric
  \end{itemize}
\end{frame}

% ================================================= RESULTS (2 frames)

\begin{frame}{Main effect estimates $2\hat\beta_i$}
  \scriptsize
  \begin{center}
  \begin{tabular}{@{}lrrrrrrr@{}}
    \toprule
    Metric & $M$ & $B$ & $S$ & $O$ & $E$ & $\hat\gamma$ & $R^2$\\
    \midrule
    Gain over default   & $+0.001$ & $-0.001$ & $+0.001^{**}$ & $+0.000$ & $+0.000$ & $-0.001^{***}$ & $0.029$\\
    Wasted trial ratio  & $-0.023^{***}$ & $-0.034^{***}$ & $+0.044^{***}$ & $-0.075^{***}$ & $+0.005$ & $-0.003$ & $0.159$\\
    Improvement rate    & $+0.008$ & $-0.026^{*}$ & $+0.034^{**}$ & $+0.006$ & $+0.018$ & $-0.064^{***}$ & $0.044$\\
    Cost to best        & $+0.06$ & $-0.02$ & $+0.05$ & $-0.01$ & $-0.11$ & $-0.20^{***}$ & $0.025$\\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.2cm}
  \small
  \begin{itemize}
    \item Reporting style is the largest effect anywhere in the design,
          $-0.075$, cutting the wasted trial ratio from $0.106$ to $0.031$
    \item The adaptive stopping rule works the other way, $+0.044$: it raises
          the mean number of experiments from $2.88$ to $4.57$, and some of the
          extra proposals duplicate earlier ones
    \item The stopping rule is the only component touching either outcome
          metric, and nothing survives on cost to best
    \item On two metrics the dataset effect $\hat\gamma$ is about twice the
          largest component effect, both at $p<10^{-5}$. That is exactly the
          variance $\gamma_j$ takes out of the residual
  \end{itemize}
  \vfill
  \fnote{HC3 errors, Benjamini and Hochberg adjusted $p$. $\hat\gamma$ is Wine
        relative to Breast Cancer. Stars: adjusted $p<0.05$, $<0.01$, $<0.001$.}
\end{frame}

\begin{frame}{What the design lets us say beyond the point estimates}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{RQ2: interactions}
        \small
        Three of forty survive adjustment, all on the wasted trial ratio, all
        involving reporting style. Only $S\times O$ at $-0.031$ is main effect
        sized, and its sign means \emph{substitution}, not reinforcement. The
        rest are additive.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Compliance, per component}
        \scriptsize
        $B$ families proposed: $0.66$ / $0.82$\\
        $S$ experiments performed: $0.89$ / $0.20$\\
        $E$ second proposal refines first: $0.88$ / $0.69$\\
        $O$ prose characters: $205$ / $386$\\
        $M$ leaves no behavioural trace
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{alertblock}{Why compliance is reported at all}
    It separates a component with no effect from one the model disregarded.
    The adaptive rule is followed in only $20\%$ of runs, so its estimate is
    the effect of \emph{offering} an adaptive budget, not of a loop that
    terminates on diminishing returns. All estimates are intention to treat,
    attenuated towards zero in proportion to non compliance.
  \end{alertblock}
\end{frame}

% ================================== IMPLICATIONS AND LIMITATIONS (2)

\begin{frame}{Implications, and why outcome barely moved}
  \begin{itemize}
    \item Within the range of wordings tested, the instruction file determines
          the \textbf{efficiency} of the search, not the \textbf{quality} of
          the configuration returned
    \item The highest leverage sentence in the file is the one about reporting
          style: requiring the model to state its reasoning cuts malformed and
          duplicate proposals by about two thirds, and costs only tokens
    \item An end to end score cannot attribute. A sliced file plus a full
          factorial can, and is not much more expensive
  \end{itemize}
  \vfill
  \begin{alertblock}{The ceiling binds, not the loop}
    Untuned defaults already reach $a_0 = 0.960$ on both tasks. $40.0\%$ of
    runs improve on that and $20.9\%$ end below it. The measured effects
    concern the efficiency with which a ceiling is approached, and say nothing
    about the height of that ceiling.
  \end{alertblock}
\end{frame}

\begin{frame}{Boundary conditions}
  \begin{enumerate}
    \item \textbf{One agent model.} What was measured is the action of
          instructions on a single policy, not a property of instructions in
          general
    \item \textbf{One wording per level.} Semantics were held fixed, phrasing
          was not varied, so variance due to wording cannot be separated from
          variance due to the architectural constraint. The most serious of the
          four
    \item \textbf{Two datasets.} The dataset fixed effect rests on a single
          contrast, and both tasks sit near the accuracy ceiling
    \item \textbf{A hyperparameter space, not code.} The instruction file
          carries less of the system's behaviour than it would in Karpathy's
          loop
  \end{enumerate}
  \vfill
  \begin{block}{The two extensions these open}
    Sample several wordings per level, which separates a component from its
    phrasing. Then apply the procedure to a loop that edits code, where
    \texttt{\prog} carries considerably more of the behaviour.
  \end{block}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item \texttt{\prog} does what controller code does in conventional
          software, yet cannot be read the way code is read, so it was treated
          as a configuration vector
    \item All $2^5=32$ files, two datasets, twenty replicates, $1{,}280$ runs
          against a real executor
    \item Because the configurations were constructed rather than collected,
          the coefficients admit a causal reading. Because the factorial is
          complete, independence could be tested rather than assumed
    \item \textbf{RQ1}: the components determine the process, not the outcome,
          and reporting style dominates
    \item \textbf{RQ2}: three of forty interactions survive, all involving
          reporting style, so the components are close to additive
  \end{itemize}
  \vfill
  \begin{block}{}
    \centering
    For tasks of this difficulty, \texttt{\prog} determines how the loop
    conducts its search rather than what the search returns.
  \end{block}
  \vfill
  \centering\small Thank you. Questions welcome.
\end{frame}
